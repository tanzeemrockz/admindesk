create table if not exists public.profiles (
  id uuid references auth.users on delete cascade primary key,
  full_name text not null,
  email text not null,
  role text default 'Developer',
  is_admin boolean default false,
  status text default 'Available',
  created_at timestamptz default now()
);

create table if not exists public.leads (
  id uuid default gen_random_uuid() primary key,
  name text not null,
  company text,
  email text,
  phone text,
  source text default 'Reddit',
  status text default 'Cold Contact',
  estimated_value_inr numeric default 0,
  follow_up_date date,
  notes text,
  created_at timestamptz default now()
);

create table if not exists public.clients (
  id uuid default gen_random_uuid() primary key,
  company_name text not null,
  contact_person text not null,
  email text,
  phone text,
  notes text,
  created_at timestamptz default now()
);

create table if not exists public.projects (
  id uuid default gen_random_uuid() primary key,
  title text not null,
  client_id uuid references public.clients(id) on delete set null,
  assigned_to_id uuid references public.profiles(id) on delete set null,
  stage text default 'Asset Intake',
  github_url text,
  vercel_url text,
  domain_url text,
  zoho_thread_url text,
  drive_url text,
  reference_urls text,
  primary_cta text default 'WhatsApp Inquiry',
  intake_checklist jsonb default '{"logo": false, "images": false, "bio": false, "catalog": false, "contact": false, "socials": false, "colors": false, "reviews": false}'::jsonb,
  fee_inr numeric default 0,
  advance_inr numeric default 0,
  balance_inr numeric generated always as (fee_inr - advance_inr) stored,
  payment_status text default 'Pending',
  deadline date,
  created_at timestamptz default now()
);

-- Upgrade existing projects table safely if already created
alter table public.projects add column if not exists drive_url text;
alter table public.projects add column if not exists reference_urls text;
alter table public.projects add column if not exists primary_cta text default 'WhatsApp Inquiry';
alter table public.projects add column if not exists intake_checklist jsonb default '{"logo": false, "images": false, "bio": false, "catalog": false, "contact": false, "socials": false, "colors": false, "reviews": false}'::jsonb;

create table if not exists public.tasks (
  id uuid default gen_random_uuid() primary key,
  project_id uuid references public.projects(id) on delete cascade,
  title text not null,
  assigned_to_id uuid references public.profiles(id) on delete set null,
  preview_url text,
  status text default 'To Do',
  created_at timestamptz default now()
);

alter table public.profiles enable row level security;

alter table public.clients enable row level security;
alter table public.projects enable row level security;
alter table public.tasks enable row level security;
